    function loadData() {
      try {
        const saved = localStorage.getItem(STORAGE_KEY);
        if (saved) {
          return JSON.parse(saved);
        }
      } catch (e) {
        console.error('Gagal memuat data:', e);
      }
      return JSON.parse(JSON.stringify(defaultData));
    }

    function saveData() {
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(appData));
        updateDashboardKPIs();

      }
    }

    function formatRupiah(num) {
      return 'Rp ' + (Number(num) || 0).toLocaleString('id-ID');
    }

    function showToast(message, type = 'success') {
      const container = document.getElementById('toast-container');
      const toast = document.createElement('div');
      const bg = type === 'error' ? 'bg-[#E86A7A]' : 'bg-[#3B8AC4]';
      toast.className = `${bg} text-white px-4 py-2.5 rounded-2xl shadow-lg text-xs font-semibold flex items-center gap-2 transition-all transform translate-y-2 opacity-0`;
      toast.innerHTML = `<span>${message}</span>`;
      container.appendChild(toast);
      requestAnimationFrame(() => {
        toast.classList.remove('translate-y-2', 'opacity-0');
      });
      setTimeout(() => {
        toast.classList.add('opacity-0', 'translate-y-2');
        setTimeout(() => toast.remove(), 300);
      }, 2800);
    }

    function openCustomPrompt({ title, message = '', defaultValue = '', isConfirmOnly = false, onConfirm }) {
      const modal = document.getElementById('custom-dialog');
      const titleEl = document.getElementById('dialog-title');
      const msgEl = document.getElementById('dialog-message');
      const inputEl = document.getElementById('dialog-input');
      const confirmBtn = document.getElementById('dialog-confirm-btn');
      const cancelBtn = document.getElementById('dialog-cancel-btn');

      titleEl.textContent = title;
      if (message) {
        msgEl.textContent = message;
        msgEl.classList.remove('hidden');
      } else {
        msgEl.classList.add('hidden');
      }

      if (isConfirmOnly) {
        inputEl.classList.add('hidden');
      } else {
        inputEl.classList.remove('hidden');
        inputEl.value = defaultValue;
      }

      modal.classList.remove('hidden');
      if (!isConfirmOnly) setTimeout(() => inputEl.focus(), 50);

      const cleanup = () => {
        modal.classList.add('hidden');
        confirmBtn.onclick = null;
        cancelBtn.onclick = null;
      };

      cancelBtn.onclick = cleanup;
      confirmBtn.onclick = () => {
        const val = inputEl.value.trim();
        cleanup();
        onConfirm(isConfirmOnly ? true : val);
      };
    }

    function updateCountdown() {
      const target = new Date(appData.couple.weddingDate).getTime();
      const now = Date.now();
      const diff = Math.max(0, target - now);

      const days = Math.floor(diff / (1000 * 60 * 60 * 24));
      const hours = Math.floor((diff / (1000 * 60 * 60)) % 24);
      const minutes = Math.floor((diff / 1000 / 60) % 60);
      const seconds = Math.floor((diff / 1000) % 60);

      document.getElementById('cd-days').textContent = days;
      document.getElementById('cd-hours').textContent = String(hours).padStart(2, '0');
      document.getElementById('cd-minutes').textContent = String(minutes).padStart(2, '0');
      document.getElementById('cd-seconds').textContent = String(seconds).padStart(2, '0');
    }

    function updateDashboardKPIs() {
      // Budget KPI
      let grandTotal = 0, paidTotal = 0;
      appData.budgetCategories.forEach(cat => {
        cat.items.forEach(item => {
          const sub = (Number(item.qty) || 0) * (Number(item.price) || 0);
          grandTotal += sub;
          if (item.status === 'Lunas') paidTotal += sub;
        });
      });
      document.getElementById('dash-budget-total').textContent = formatRupiah(grandTotal);
      document.getElementById('dash-budget-sub').textContent = `Lunas: ${formatRupiah(paidTotal)}`;

      // Guest KPI
      const totalGuests = appData.guests.length;
      const attendingGuests = appData.guests.filter(g => g.rsvp === 'Hadir').length;
      document.getElementById('dash-guest-total').textContent = `${totalGuests} Tamu`;
      document.getElementById('dash-guest-sub').textContent = `${attendingGuests} Konfirmasi Hadir`;

      // Checklist KPI
      let totalTasks = 0, doneTasks = 0;
      appData.timelinePhases.forEach(ph => {
        totalTasks += ph.tasks.length;
        doneTasks += ph.tasks.filter(t => t.done).length;
      });
      const pct = totalTasks === 0 ? 0 : Math.round((doneTasks / totalTasks) * 100);
      document.getElementById('dash-task-percent').textContent = `${pct}% (${doneTasks}/${totalTasks})`;
      document.getElementById('dash-task-bar').style.width = `${pct}%`;

      // Moodboard KPI
      const totalCols = appData.moodboard.collections.length;
      const totalMoods = appData.moodboard.collections.reduce((sum, c) => sum + c.items.length, 0);
      document.getElementById('dash-mood-total').textContent = `${totalCols} Koleksi`;
      document.getElementById('dash-mood-sub').textContent = `${totalMoods} Inspirasi tersimpan`;
    }

    function addBudgetCategory() {
      openCustomPrompt({
        title: 'Tambah Part / Kategori Anggaran',
        defaultValue: 'Kategori Baru (Misal: Souvenir)',
        onConfirm: (val) => {
          if (!val) return;
          appData.budgetCategories.push({
            id: 'cat-' + Date.now(),
            title: val,
            items: [
              { id: 'itm-' + Date.now(), name: 'Item Pertama', vendor: 'Nama Toko', qty: 1, price: 0, status: 'Booking' }
            ]
          });
          saveData();
          renderBudget();
          showToast(`Part "${val}" ditambahkan!`);
        }
      });
    }

    function renameBudgetCategory(catId, newTitle) {
      const cat = appData.budgetCategories.find(c => c.id === catId);
      if (cat && newTitle.trim()) {
        cat.title = newTitle.trim();
        saveData();
        renderBudget();
        showToast('Judul part berhasil diperbarui');
      }
    }

    function deleteBudgetCategory(catId) {
      openCustomPrompt({
        title: 'Hapus Part Anggaran?',
        message: 'Semua daftar barang di dalam kategori ini akan ikut terhapus.',
        isConfirmOnly: true,
        onConfirm: () => {
          appData.budgetCategories = appData.budgetCategories.filter(c => c.id !== catId);
          saveData();
          renderBudget();
          showToast('Kategori dihapus');
        }
      });
    }

    function addBudgetItem(catId) {
      const cat = appData.budgetCategories.find(c => c.id === catId);
      if (!cat) return;
      cat.items.push({
        id: 'itm-' + Date.now(),
        name: '',
        vendor: '',
        qty: 1,
        price: 0,
        status: 'Booking'
      });
      saveData();
      renderBudget();
    }

    function updateBudgetItem(catId, itemId, field, value) {
      const cat = appData.budgetCategories.find(c => c.id === catId);
      if (!cat) return;
      const item = cat.items.find(i => i.id === itemId);
      if (!item) return;
      item[field] = value;
      saveData();
      renderBudget();
    }

    function deleteBudgetItem(catId, itemId) {
      const cat = appData.budgetCategories.find(c => c.id === catId);
      if (!cat) return;
      cat.items = cat.items.filter(i => i.id !== itemId);
      saveData();
      renderBudget();
    }

    function renderGuests() {
      const search = document.getElementById('guest-search').value.toLowerCase();
      const filterType = document.getElementById('guest-filter-type').value;
      const filterStatus = document.getElementById('guest-filter-status').value;
      const filterRsvp = document.getElementById('guest-filter-rsvp').value;

      // Update Counters
      const all = appData.guests;
      document.getElementById('guest-stat-total').textContent = `Total: ${all.length}`;
      document.getElementById('guest-stat-cetak').textContent = `Cetak: ${all.filter(g => g.type === 'Cetak').length}`;
      document.getElementById('guest-stat-digital').textContent = `Digital: ${all.filter(g => g.type === 'Digital').length}`;
      document.getElementById('guest-stat-hadir').textContent = `Hadir: ${all.filter(g => g.rsvp === 'Hadir').length}`;
      document.getElementById('guest-stat-tidak').textContent = `Tidak: ${all.filter(g => g.rsvp === 'Tidak').length}`;

      const filtered = all.filter(g => {
        const matchName = g.name.toLowerCase().includes(search);
        const matchType = filterType === 'ALL' || g.type === filterType;
        const matchStatus = filterStatus === 'ALL' || g.status === filterStatus;
        const matchRsvp = filterRsvp === 'ALL' || g.rsvp === filterRsvp;
        return matchName && matchType && matchStatus && matchRsvp;
      });

      const tbody = document.getElementById('guest-table-body');
      if (filtered.length === 0) {
        tbody.innerHTML = `<tr><td colspan="6" class="py-8 text-center text-slate-400 text-sm">Belum ada data tamu yang cocok.</td></tr>`;
        return;
      }

      tbody.innerHTML = filtered.map((guest, index) => {
        const rsvpStyle = guest.rsvp === 'Hadir'
          ? 'bg-emerald-50 text-[#6E9E47] border-emerald-200'
          : guest.rsvp === 'Tidak'
            ? 'bg-rose-50 text-[#E86A7A] border-rose-200'
            : 'bg-slate-50 text-slate-600 border-slate-200';

          ? 'bg-sky-50 text-[#3B8AC4] border-sky-200'

          <tr class="hover:bg-sky-50/30 transition">
            <td class="py-2.5 px-4 text-center font-semibold text-slate-400">${index + 1}</td>
            <td class="py-2 px-4">
              <input type="text" value="${escapeHtml(guest.name)}"
                onchange="updateGuest('${guest.id}', 'name', this.value)"
                class="w-full px-2 py-1 rounded-lg bg-transparent hover:bg-slate-50 focus:bg-white focus:ring-1 focus:ring-[#62B4E8] focus:outline-none font-medium text-slate-800" />
            </td>
            <td class="py-2 px-4">
              <select onchange="updateGuest('${guest.id}', 'type', this.value)"
                class="w-full px-2.5 py-1.5 rounded-xl text-xs font-semibold border border-slate-200 bg-white text-slate-700 cursor-pointer">
                <option value="Cetak" ${guest.type === 'Cetak' ? 'selected' : ''}>Cetak</option>
                <option value="Digital" ${guest.type === 'Digital' ? 'selected' : ''}>Digital</option>
              </select>
            </td>
            <td class="py-2 px-4">
              <select onchange="updateGuest('${guest.id}', 'status', this.value)"
                class="w-full px-2.5 py-1.5 rounded-xl text-xs font-bold border cursor-pointer ${statusStyle}">
                <option value="Belum Diundang" ${guest.status === 'Belum Diundang' ? 'selected' : ''}>Belum Diundang</option>
                <option value="Sudah Diundang" ${guest.status === 'Sudah Diundang' ? 'selected' : ''}>Sudah Diundang</option>
              </select>
            </td>
            <td class="py-2 px-4">
              <select onchange="updateGuest('${guest.id}', 'rsvp', this.value)"
                class="w-full px-2.5 py-1.5 rounded-xl text-xs font-bold border cursor-pointer ${rsvpStyle}">
                <option value="Belum Konfirmasi" ${guest.rsvp === 'Belum Konfirmasi' ? 'selected' : ''}>Belum Konfirmasi</option>
                <option value="Hadir" ${guest.rsvp === 'Hadir' ? 'selected' : ''}>Hadir</option>
                <option value="Tidak" ${guest.rsvp === 'Tidak' ? 'selected' : ''}>Tidak</option>
              </select>
            </td>
            <td class="py-2 px-4 text-center">
              <button onclick="deleteGuest('${guest.id}')" title="Hapus Tamu" class="p-1.5 rounded-lg text-slate-400 hover:text-[#E86A7A] hover:bg-rose-50 transition">
                <i data-lucide="trash-2" class="w-4 h-4"></i>
              </button>

    function toggleTask(phaseId, taskId) {
      const ph = appData.timelinePhases.find(p => p.id === phaseId);
      if (!ph) return;
      const t = ph.tasks.find(item => item.id === taskId);
      if (t) {
        t.done = !t.done;
        saveData();
        renderChecklist();
      }
    }

    function updateTask(phaseId, taskId, field, value) {
      const ph = appData.timelinePhases.find(p => p.id === phaseId);
      if (!ph) return;
      const t = ph.tasks.find(item => item.id === taskId);
      if (t) {
        t[field] = value;
        saveData();
        renderChecklist();
      }
    }

      if (!ph) return;

      saveData();
      renderChecklist();

    // ==================== IMAGE COMPRESSION & BACKUP/RESTORE ====================
    function compressImage(file, maxWidth, callback) {
      const reader = new FileReader();
      reader.onload = (e) => {
        const img = new Image();
        img.onload = () => {
          const canvas = document.createElement('canvas');
          let w = img.width;
          let h = img.height;
          if (w > maxWidth) {
            h = Math.round((h * maxWidth) / w);
            w = maxWidth;
          }
          canvas.width = w;
          canvas.height = h;
          const ctx = canvas.getContext('2d');
          ctx.drawImage(img, 0, 0, w, h);
          callback(canvas.toDataURL('image/jpeg', 0.78));
        };
        img.src = e.target.result;
      };
      reader.readAsDataURL(file);
    }

    function exportData() {
      const dataStr = JSON.stringify(appData, null, 2);
      const blob = new Blob([dataStr], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `wedding-planner-${appData.couple.bride}-${appData.couple.groom}.json`.toLowerCase().replace(/\s+/g, '-');
      a.click();
      URL.revokeObjectURL(url);
      showToast('File backup berhasil diunduh!');
    }

    function importData(event) {
      const file = event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = (e) => {
        try {
          const parsed = JSON.parse(e.target.result);
          if (parsed && parsed.couple && parsed.budgetCategories) {
            appData = parsed;
            saveData();
            initApp();
            showToast('Data berhasil dipulihkan!');
          } else {
            showToast('Format file backup tidak valid.', 'error');
          }
        } catch (err) {
          showToast('Gagal membaca file JSON.', 'error');
        }
      };
      reader.readAsText(file);
    }

    function escapeHtml(str) {
      if (str === null || str === undefined) return '';
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;');
    }
